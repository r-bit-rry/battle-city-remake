# P2 Ally AI v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the parameter-scaled ally AI with a two-layer utility brain (strategic goal scoring + per-frame tactics), four capability-gated tiers, three selectable roles, and a polite coordination model, verified by scenario tests and seeded soak benchmarks.

**Architecture:** `aiFrameLogic(game)` builds a pure snapshot each frame, computes threat scores and a predicted-bullet danger map, re-scores six goals every `decisionInterval` frames with role-weighted utilities and hysteresis, then executes the current goal per frame through tactics (dodge → fire → brick drill → movement). All modules under `src/engine/logics/ai/` are pure and unit-testable; only `ai.logic.ts` touches `Game`.

**Tech Stack:** TypeScript, Vitest 2 (`npm test`), Vite build, ESLint + Prettier.

## Global Constraints

- Formatting: tabs, single quotes, no semicolons, trailing comma `es5` (`.prettierrc`).
- No code comments unless the task explicitly includes them.
- No new runtime dependencies.
- Keep the public `aiFrameLogic(game: Game): void` signature and the `game.aiLevel` plumbing.
- Internal tier keys stay `beginner | intermediate | pro | elite`; display labels stay `0..3`.
- Role type name: `TypeAiRole = 'guardian' | 'wingman' | 'hunter'`; default role `wingman`.
- Direction semantics: `TOP` = `y + 1`, `BOTTOM` = `y - 1`, `LEFT` = `x - 1`, `RIGHT` = `x + 1`.
- Lanes: coordinate values `v` in `7..200` with `v % 8 === 0 || v % 8 === 7`.
- Base/flag center: `(103, 7)`; guard point: `(103, 31)`.
- Test helpers: engine tests use `// @vitest-environment node` and seed `Math.random` with the LCG from `src/engine/engine.test.ts`.
- Commands: `npm test`, `npm run build`, `npm run lint`.
- Spec: `docs/superpowers/specs/2026-09-24-ai-ally-v2-design.md`.

```ts
import { Game } from '../../instances'
import { isEnemy, isImpassibleGameObject, isPlayerDead } from '../../utils'
import { TypeEnemyTank } from '../../types'
import {
	AiBonusSnapshot,
	AiBulletSnapshot,
	AiEnemySnapshot,
	AiMemory,
	AiObstacleSnapshot,
	AiPlayerSnapshot,
	AiSnapshot,
	PlayerIntent,
} from './ai.types'

export const FLAG_X = 103
export const FLAG_Y = 7
export const GUARD_Y = 31
export const PLAYER_TRACK_SIZE = 60

export const createMemory = (): AiMemory => ({
	goal: null,
	goalScore: 0,
	cooldown: 0,
	fireCooldown: 0,
	path: null,
	pathSignature: null,
	lastX: -1,
	lastY: -1,
	lastMove: null,
	stuckFrames: 0,
	stuckRecoveries: 0,
	playerTrack: [],
	lastTargetId: null,
})

export const inferPlayerIntent = (
	track: { x: number; y: number }[],
	player: AiPlayerSnapshot
): PlayerIntent => {
	if (player.dead || player.lives === 0) return 'dead'
	if (player.lives === 1) return 'retreating'
	if (track.length < 2) return 'holding'

	const dy = track[track.length - 1].y - track[0].y
	if (dy > 16) return 'pushing'
	if (dy < -16) return 'retreating'

	return 'holding'
}

const mapSignature = (game: Game) => {
	let hash = game.isFlagAlive ? 1 : 0
	for (const object of game.objects) {
		hash = (hash * 31 + object.coordinateX * 8 + object.coordinateY) | 0
	}

	return String(hash)
}

export const buildSnapshot = (game: Game, memory: AiMemory): AiSnapshot => {
	const ai = game.p2
	if (!ai) throw new Error('AI snapshot requires p2')

	memory.playerTrack.push({
		x: game.p1.coordinateX,
		y: game.p1.coordinateY,
	})
	if (memory.playerTrack.length > PLAYER_TRACK_SIZE)
		memory.playerTrack.shift()

	const player: AiPlayerSnapshot = {
		x: game.p1.coordinateX,
		y: game.p1.coordinateY,
		direction: game.p1.direction,
		lives: game.p1.lives,
		dead: isPlayerDead(game.p1),
	}

	const enemies: AiEnemySnapshot[] = game.enemies
		.filter((enemy) => !enemy.spawnAnimation)
		.map((enemy) => ({
			id: enemy.id,
			x: enemy.coordinateX,
			y: enemy.coordinateY,
			direction: enemy.direction,
			type: enemy.type as TypeEnemyTank,
			speed: enemy.type === 'SPEEDY' ? 2 : 1,
			carrier: Boolean(enemy.bonus),
			lives: enemy.lives,
			availableBullets: enemy.availableBullets,
		}))

	const bullets: AiBulletSnapshot[] = game.bullets.map((bullet) => ({
		id: bullet.id,
		x: bullet.coordinateX,
		y: bullet.coordinateY,
		direction: bullet.direction,
		speed: bullet.speed,
```

```ts
import { TypeDirection } from '../../types'
import { FLAG_X, FLAG_Y } from './ai.perception'
import {
	AiBulletSnapshot,
	AiEnemySnapshot,
	AiSnapshot,
	DangerMap,
	Threat,
} from './ai.types'

const LANE_VALUES: number[] = []
for (let value = 7; value <= 200; value++) {
	if (value % 8 === 0 || value % 8 === 7) LANE_VALUES.push(value)
}

const distance = (x1: number, y1: number, x2: number, y2: number) =>
	Math.hypot(x1 - x2, y1 - y2)

export const nearestLane = (value: number) =>
	LANE_VALUES.reduce(
		(best, lane) => (Math.abs(lane - value) < Math.abs(best - value) ? lane : best),
		LANE_VALUES[0]
	)

export const advance = (
	x: number,
	y: number,
	direction: TypeDirection,
	amount: number
) => {
	switch (direction) {
		case 'TOP':
			return { x, y: y + amount }
		case 'BOTTOM':
			return { x, y: y - amount }
		case 'LEFT':
			return { x: x - amount, y }
		case 'RIGHT':
			return { x: x + amount, y }
	}
}

const typeWeight = (type: AiEnemySnapshot['type']) =>
	type === 'SPEEDY' ? 1 : type === 'HEAVY' ? 0.6 : 0.5

const obstacleAt = (snapshot: AiSnapshot, x: number, y: number) =>
	snapshot.obstacles.find(
		(object) =>
			object.blocksBullets &&
			x >= object.x - 2 &&
			x <= object.x + 9 &&
			y >= object.y - 2 &&
			y <= object.y + 9
	)

export const hasLineToBase = (
	snapshot: AiSnapshot,
	enemy: AiEnemySnapshot
) => {
	const steps = Math.ceil(
		distance(enemy.x, enemy.y, FLAG_X, FLAG_Y) / 4
	)
	for (let i = 1; i < steps; i++) {
		const t = i / steps
		const x = enemy.x + (FLAG_X - enemy.x) * t
		const y = enemy.y + (FLAG_Y - enemy.y) * t
		if (obstacleAt(snapshot, x, y)) return false
	}

	return true
}

const aimingPlayer = (snapshot: AiSnapshot, enemy: AiEnemySnapshot) => {
	const dx = snapshot.player.x - enemy.x
	const dy = snapshot.player.y - enemy.y
	if (Math.abs(dx) > 7 && Math.abs(dy) > 7) return false

	const facing =
		(enemy.direction === 'TOP' && dy > 0) ||
		(enemy.direction === 'BOTTOM' && dy < 0) ||
		(enemy.direction === 'LEFT' && dx < 0) ||
		(enemy.direction === 'RIGHT' && dx > 0)

	return (
		facing && distance(enemy.x, enemy.y, snapshot.player.x, snapshot.player.y) <= 120
	)
}

export const computeThreats = (snapshot: AiSnapshot): Threat[] =>
	snapshot.enemies
		.map((enemy) => {
			const distToBase = distance(enemy.x, enemy.y, FLAG_X, FLAG_Y)
			const line = hasLineToBase(snapshot, enemy)
			const aiming = aimingPlayer(snapshot, enemy)
			const score =
				0.4 * (1 - Math.min(1, distToBase / 200)) +
				0.2 * typeWeight(enemy.type) +
				0.2 * (line ? 1 : 0) +
				0.1 * (enemy.carrier ? 1 : 0) +
				0.1 * (aiming ? 1 : 0)

			return {
				enemyId: enemy.id,
				score,
				hasLineToBase: line,
				aimingPlayer: aiming,
			}
		})
		.sort((a, b) => b.score - a.score)

export const predictedHit = (
	bullet: AiBulletSnapshot,
	snapshot: AiSnapshot,
	x: number,
	y: number,
	horizon: number
) => {
	if (horizon <= 0) return false
	let bx = bullet.x
	let by = bullet.y
	for (let t = 1; t <= horizon; t++) {
		const next = advance(bx, by, bullet.direction, bullet.speed)
		bx = next.x
		by = next.y
		if (obstacleAt(snapshot, bx, by)) return false
		if (Math.abs(bx - x) <= 7 && Math.abs(by - y) <= 7) return true
	}

	return false
}

export const findIncomingBullet = (
	snapshot: AiSnapshot,
	horizon: number
): AiBulletSnapshot | null => {
	if (horizon <= 0) return null
	let best: AiBulletSnapshot | null = null
	let bestFrame = Infinity
	for (const bullet of snapshot.bullets) {
		if (!bullet.enemy) continue
		let bx = bullet.x
		let by = bullet.y
		for (let t = 1; t <= horizon; t++) {
			const next = advance(bx, by, bullet.direction, bullet.speed)
			bx = next.x
			by = next.y
			if (obstacleAt(snapshot, bx, by)) break
			if (
				Math.abs(bx - snapshot.ai.x) <= 7 &&
				Math.abs(by - snapshot.ai.y) <= 7
			) {
				if (t < bestFrame) {
					bestFrame = t
					best = bullet
				}
				break
			}
		}
	}

	return best
}

export const buildDangerMap = (
	snapshot: AiSnapshot,
	horizon: number
): DangerMap => {
	const cells = new Map<string, number>()
	for (const bullet of snapshot.bullets) {
		if (!bullet.enemy) continue
		let bx = bullet.x
		let by = bullet.y
		for (let t = 1; t <= horizon; t++) {
			const next = advance(bx, by, bullet.direction, bullet.speed)
			bx = next.x
			by = next.y
			if (obstacleAt(snapshot, bx, by)) break
			const key = `${nearestLane(bx)},${nearestLane(by)}`
			const current = cells.get(key)
			if (current === undefined || current > t) cells.set(key, t)
		}
	}

	return { cells }
}

export const isCellDangerous = (
	danger: DangerMap,
	x: number,
	y: number,
	withinFrames: number
) => {
	const frame = danger.cells.get(`${x},${y}`)

	return frame !== undefined && frame <= withinFrames
}
```

```ts
import { BusyCoordinates } from '../../types'
import { isImpassibleGameObject, isTank } from '../../utils'
import { DangerMap, LanePath } from './ai.types'

export const LANES: number[] = []
for (let value = 7; value <= 200; value++) {
	if (value % 8 === 0 || value % 8 === 7) LANES.push(value)
}
export const LANE_COUNT = LANES.length

export interface PathOptions {
	allowBrick: boolean
	danger?: DangerMap
	dangerPenalty?: number
	dangerHorizon?: number
	ignoreId?: string
}

interface CellInfo {
	cost: number
	brick: boolean
}

const nearestLaneIndex = (value: number) => {
	let best = 0
	for (let i = 1; i < LANE_COUNT; i++) {
		if (Math.abs(LANES[i] - value) < Math.abs(LANES[best] - value)) best = i
	}

	return best
}

const blockersAt = (
	x: number,
	y: number,
	busy: BusyCoordinates[],
	ignoreId?: string
) => {
	let hardBlocked = false
	const bricks: BusyCoordinates[] = []
	for (const object of busy) {
		if (object.id === ignoreId) continue
		if (!isTank(object.type) && !isImpassibleGameObject(object.type)) continue

		const minX = isTank(object.type) ? object.coordinateX - 7 : object.coordinateX
		const minY = isTank(object.type) ? object.coordinateY - 7 : object.coordinateY
		const maxX = object.coordinateX + 7
		const maxY = object.coordinateY + 7

		if (x - 7 <= maxX && minX <= x + 7 && y - 7 <= maxY && minY <= y + 7) {
			if (object.type === 'BRICK') bricks.push(object)
			else hardBlocked = true
		}
	}

	return { hardBlocked, bricks }
}

const cellInfo = (
	x: number,
	y: number,
	busy: BusyCoordinates[],
	options: PathOptions
): CellInfo | null => {
	if (x < 7 || x > 200 || y < 7 || y > 200) return null

	const { hardBlocked, bricks } = blockersAt(x, y, busy, options.ignoreId)
	if (hardBlocked) return null
	if (bricks.length) {
		if (!options.allowBrick) return null

		return { cost: 3, brick: true }
	}

	let cost = 1
	const frame = options.danger?.cells.get(`${x},${y}`)
	if (frame !== undefined && options.dangerPenalty) {
		const horizon = options.dangerHorizon ?? 30
		cost += options.dangerPenalty * Math.max(0, 1 - frame / horizon)
	}

	return { cost, brick: false }
}

const heuristic = (a: number, b: number) => {
	const ax = a % LANE_COUNT
	const ay = (a - ax) / LANE_COUNT
	const bx = b % LANE_COUNT
	const by = (b - bx) / LANE_COUNT

	return (Math.abs(ax - bx) + Math.abs(ay - by)) / 8
}

class MinHeap {
	private items: { node: number; priority: number }[] = []

	get size() {
		return this.items.length
	}

	push(node: number, priority: number) {
		const items = this.items
		items.push({ node, priority })
		let index = items.length - 1
		while (index > 0) {
			const parent = (index - 1) >> 1
			if (items[parent].priority <= items[index].priority) break
			;[items[parent], items[index]] = [items[index], items[parent]]
			index = parent
		}
	}

	pop() {
		const items = this.items
		const top = items[0]
		const last = items.pop() as { node: number; priority: number }
		if (items.length) {
			items[0] = last
			let index = 0
			for (;;) {
				const left = index * 2 + 1
				const right = left + 1
				let smallest = index
				if (
					left < items.length &&
					items[left].priority < items[smallest].priority
				)
					smallest = left
				if (
					right < items.length &&
					items[right].priority < items[smallest].priority
				)
					smallest = right
				if (smallest === index) break
				;[items[smallest], items[index]] = [items[index], items[smallest]]
				index = smallest
			}
		}

		return top.node
	}
}

const NEIGHBORS = [
	[1, 0],
	[-1, 0],
	[0, 1],
	[0, -1],
]

export const findPath = (
	tank: { id: string; coordinateX: number; coordinateY: number },
	targetX: number,
	targetY: number,
	busy: BusyCoordinates[],
	options: PathOptions
): LanePath | null => {
	const startIndex =
		nearestLaneIndex(tank.coordinateY) * LANE_COUNT +
		nearestLaneIndex(tank.coordinateX)
	const goalIndex =
		nearestLaneIndex(targetY) * LANE_COUNT + nearestLaneIndex(targetX)
	if (startIndex === goalIndex) return { waypoints: [], brick: null }

	const total = LANE_COUNT * LANE_COUNT
	const cost = new Float64Array(total).fill(Infinity)
	const parents = new Int32Array(total).fill(-1)
	const closed = new Uint8Array(total)
	const open = new MinHeap()

	cost[startIndex] = 0
	open.push(startIndex, heuristic(startIndex, goalIndex))

	while (open.size) {
		const current = open.pop()
		if (closed[current]) continue
		closed[current] = 1
		if (current === goalIndex) break

		const xi = current % LANE_COUNT
		const yi = (current - xi) / LANE_COUNT

		for (const [dx, dy] of NEIGHBORS) {
			const nx = xi + dx
			const ny = yi + dy
			if (nx < 0 || nx >= LANE_COUNT || ny < 0 || ny >= LANE_COUNT)
				continue

			const next = ny * LANE_COUNT + nx
			const info = cellInfo(LANES[nx], LANES[ny], busy, options)
			if (!info) continue

			const tentative = cost[current] + info.cost
			if (tentative < cost[next]) {
				cost[next] = tentative
				parents[next] = current
				open.push(next, tentative + heuristic(next, goalIndex))
			}
		}
	}

	if (parents[goalIndex] === -1) return null

	const waypoints: { x: number; y: number }[] = []
	let node = goalIndex
	while (node !== startIndex) {
		const xi = node % LANE_COUNT
		const yi = (node - xi) / LANE_COUNT
		waypoints.push({ x: LANES[xi], y: LANES[yi] })
		node = parents[node]
	}
	waypoints.reverse()

	let brick: { x: number; y: number } | null = null
	if (options.allowBrick) {
		let cursor = goalIndex
		while (cursor !== startIndex) {
			const xi = cursor % LANE_COUNT
			const yi = (cursor - xi) / LANE_COUNT
			const info = cellInfo(LANES[xi], LANES[yi], busy, options)
			if (info?.brick) {
				brick = { x: LANES[xi], y: LANES[yi] }
				break
			}
			cursor = parents[cursor]
		}
	}

	return { waypoints, brick }
}
```

```ts
import {
	AiBonusSnapshot,
	AiGoal,
	AiMemory,
	AiRoleProfile,
	AiSnapshot,
	AiTierFeatures,
	Threat,
} from './ai.types'
import { FLAG_X, FLAG_Y, GUARD_Y } from './ai.perception'
import { retreatLives } from './ai.profiles'

const distance = (x1: number, y1: number, x2: number, y2: number) =>
	Math.hypot(x1 - x2, y1 - y2)

const enemyById = (snapshot: AiSnapshot, id: string) =>
	snapshot.enemies.find((enemy) => enemy.id === id)

export const isGoalValid = (goal: AiGoal, snapshot: AiSnapshot) => {
	if (goal.enemyId && !snapshot.enemies.some((e) => e.id === goal.enemyId))
		return false
	if (goal.bonusId && !snapshot.bonuses.some((b) => b.id === goal.bonusId))
		return false

	return true
}

export const bonusValue = (
	snapshot: AiSnapshot,
	bonus: AiBonusSnapshot,
	features: AiTierFeatures,
	threats: Threat[]
): number => {
	if (features.bonusLogic === 'none') return 0
	if (features.bonusLogic === 'safe') return 1

	switch (bonus.type) {
		case 'STAR':
			return snapshot.ai.type === 'LVL_0'
				? 1
				: snapshot.ai.type === 'LVL_1'
					? 0.8
					: 0.4
		case 'HP':
			return snapshot.ai.lives < 3 ? 1 : 0.2
		case 'GRENADE':
			return snapshot.enemies.length >= 2 ? 1 : 0.3
		case 'HELMET': {
			const nearBase = threats.some((threat) => {
				const enemy = enemyById(snapshot, threat.enemyId)
				return (
					enemy &&
					distance(enemy.x, enemy.y, FLAG_X, FLAG_Y) <=
						features.baseDangerRadius
				)
			})
			return nearBase ? 0.9 : 0.5
		}
		case 'TIMER':
			return snapshot.enemies.length >= 2 ? 0.9 : 0.4
	}
}

const defendGoal = (
	snapshot: AiSnapshot,
	threats: Threat[],
	features: AiTierFeatures
): { goal: AiGoal; score: number } | null => {
	for (const threat of threats) {
		const enemy = enemyById(snapshot, threat.enemyId)
		if (!enemy) continue
		if (distance(enemy.x, enemy.y, FLAG_X, FLAG_Y) > features.baseDangerRadius)
			continue

		let target = { x: enemy.x, y: enemy.y }
		if (features.intercept && enemy.direction === 'BOTTOM' && enemy.y > GUARD_Y)
			target = { x: enemy.x, y: Math.max(GUARD_Y, enemy.y - 32) }

		return {
			goal: { kind: 'defend', x: target.x, y: target.y, enemyId: enemy.id },
			score: threat.score,
		}
	}

	return null
}

const engageGoal = (
	snapshot: AiSnapshot,
	features: AiTierFeatures
): { goal: AiGoal; score: number } | null => {
	const candidates = snapshot.enemies
		.filter(
			(enemy) =>
				distance(enemy.x, enemy.y, snapshot.ai.x, snapshot.ai.y) <=
				features.leashRange
		)
		.map((enemy) => {
			const playerNear =
				!snapshot.player.dead &&
				distance(enemy.x, enemy.y, snapshot.player.x, snapshot.player.y) < 24
			const score =
				0.6 *
					(1 -
						Math.min(
							1,
							distance(enemy.x, enemy.y, snapshot.ai.x, snapshot.ai.y) / 200
						)) +
				0.25 * (1 - enemy.lives / 3) +
				0.15 * (enemy.carrier ? 1 : 0) -
				(playerNear ? 0.1 : 0)

			return { enemy, score }
		})
		.sort((a, b) => b.score - a.score)

	const best = candidates[0]
	if (!best) return null

	return {
		goal: {
			kind: 'engage',
			x: best.enemy.x,
			y: best.enemy.y,
			enemyId: best.enemy.id,
		},
		score: Math.max(0, best.score),
	}
}

const collectGoal = (
	snapshot: AiSnapshot,
	features: AiTierFeatures,
	threats: Threat[]
): { goal: AiGoal; score: number } | null => {
	if (features.bonusLogic === 'none') return null

	const valued = snapshot.bonuses
		.filter(
			(bonus) =>
				!snapshot.enemies.some(
					(enemy) => distance(enemy.x, enemy.y, bonus.x, bonus.y) <= 40
				)
		)
		.map((bonus) => {
			let value = bonusValue(snapshot, bonus, features, threats)
			if (
				features.yieldBonuses &&
				!snapshot.player.dead &&
				distance(snapshot.player.x, snapshot.player.y, bonus.x, bonus.y) <
					distance(snapshot.ai.x, snapshot.ai.y, bonus.x, bonus.y)
			)
				value = 0

			return {
				bonus,
				value,
				score:
					value *
					(1 -
						Math.min(
							1,
							distance(snapshot.ai.x, snapshot.ai.y, bonus.x, bonus.y) / 300
						)),
			}
		})
		.filter((candidate) => candidate.value > 0)
		.sort((a, b) => b.score - a.score)

	const best = valued[0]
	if (!best) return null

	return {
		goal: {
			kind: 'collect',
			x: best.bonus.x,
			y: best.bonus.y,
			bonusId: best.bonus.id,
		},
		score: best.score,
	}
}

const supportGoal = (
	snapshot: AiSnapshot,
	features: AiTierFeatures
): { goal: AiGoal; score: number } | null => {
	if (!features.coordination || snapshot.playerIntent === 'dead') return null

	if (snapshot.playerIntent === 'retreating') {
		const pursuer = snapshot.enemies
			.map((enemy) => ({
				enemy,
				distance: distance(
					enemy.x,
					enemy.y,
					snapshot.player.x,
					snapshot.player.y
				),
			}))
			.sort((a, b) => a.distance - b.distance)[0]?.enemy
		const goal: AiGoal = pursuer
			? {
					kind: 'support',
					x: (snapshot.player.x + pursuer.x) / 2,
					y: (snapshot.player.y + pursuer.y) / 2,
				}
			: { kind: 'support', x: snapshot.player.x, y: snapshot.player.y }

		return { goal, score: 1.1 }
	}

	if (snapshot.playerIntent === 'pushing') {
		return {
			goal: {
				kind: 'support',
				x: Math.min(166, Math.max(40, snapshot.player.x)),
				y: GUARD_Y + 24,
			},
			score: 0.8,
		}
	}

	return {
		goal: {
			kind: 'support',
			x: snapshot.player.x < FLAG_X ? 160 : 46,
			y: 110,
		},
		score: 0.7,
	}
}

const retreatGoal = (
	snapshot: AiSnapshot,
	features: AiTierFeatures,
	role: AiRoleProfile
): { goal: AiGoal; score: number } | null => {
	const nearby = snapshot.enemies.filter(
		(enemy) =>
			distance(enemy.x, enemy.y, snapshot.ai.x, snapshot.ai.y) <= 80
	)
	if (snapshot.ai.lives > retreatLives(features, role)) return null
	if (nearby.length < 2) return null

	return {
		goal: { kind: 'retreat', x: FLAG_X, y: GUARD_Y },
		score: 0.85,
	}
}

const patrolGoal = (
	snapshot: AiSnapshot,
	role: AiRoleProfile
): { goal: AiGoal; score: number } | null => {
	const anchor =
		role.patrolAnchor === 'base' || snapshot.player.dead
			? { x: FLAG_X, y: GUARD_Y }
			: role.patrolAnchor === 'middle'
				? { x: FLAG_X, y: 110 }
				: { x: snapshot.player.x, y: snapshot.player.y }
	const dist = distance(snapshot.ai.x, snapshot.ai.y, anchor.x, anchor.y)
	if (dist <= 28) return null

	return { goal: { kind: 'patrol', x: anchor.x, y: anchor.y }, score: 0.3 }
}

export const selectGoal = (
	snapshot: AiSnapshot,
	threats: Threat[],
	features: AiTierFeatures,
	role: AiRoleProfile,
	memory: AiMemory
): AiGoal | null => {
	const candidates = [
		defendGoal(snapshot, threats, features),
		engageGoal(snapshot, features),
		collectGoal(snapshot, features, threats),
		supportGoal(snapshot, features),
		retreatGoal(snapshot, features, role),
		patrolGoal(snapshot, role),
	].filter((candidate): candidate is { goal: AiGoal; score: number } =>
		Boolean(candidate)
	)

	const best = candidates
		.map((candidate) => ({
			goal: candidate.goal,
			weighted: candidate.score * role.weights[candidate.goal.kind],
		}))
		.sort((a, b) => b.weighted - a.weighted)[0]

	if (!best) return null

	const current = memory.goal
	if (
		current &&
		isGoalValid(current, snapshot) &&
		best.weighted <= memory.goalScore * 1.15
	)
		return current

	memory.goalScore = best.weighted
	if (best.goal.enemyId) memory.lastTargetId = best.goal.enemyId

	return best.goal
}
```

```ts
import { isTankBlockedAt } from '../../utils'
import {
	AiDecision,
	AiGoal,
	AiMemory,
	AiRoleProfile,

	AiTierFeatures,
	DangerMap,
	LanePath,
} from './ai.types'

export const ALL_DIRECTIONS: TypeDirection[] = [
	'TOP',
	'BOTTOM',
	'LEFT',
	'RIGHT',
]

const WAYPOINT_REACHED_DISTANCE = 8

const distance = (x1: number, y1: number, x2: number, y2: number) =>
	Math.hypot(x1 - x2, y1 - y2)

export const stepPosition = (
	x: number,
	y: number,
	direction: TypeDirection,
	amount = 1
) => {
	switch (direction) {
		case 'TOP':
			return { x, y: y + amount }
		case 'BOTTOM':
			return { x, y: y - amount }
		case 'LEFT':
			return { x: x - amount, y }
		case 'RIGHT':
			return { x: x + amount, y }
	}
}

export const canStep = (
	ai: { id: string; x: number; y: number },
	direction: TypeDirection,
	busy: BusyCoordinates[]
) => {
	const next = stepPosition(ai.x, ai.y, direction, 1)

	return !isTankBlockedAt({ id: ai.id }, next.x, next.y, busy)
}

export const directionToWaypoint = (
	ai: { x: number; y: number },
	waypoint: { x: number; y: number }
): TypeDirection | null => {
	if (waypoint.x !== ai.x) return waypoint.x > ai.x ? 'RIGHT' : 'LEFT'
	if (waypoint.y !== ai.y) return waypoint.y > ai.y ? 'TOP' : 'BOTTOM'

	return null
}

const isCellSafe = (
	snapshot: AiSnapshot,
	x: number,
	y: number,
	horizon: number
) =>
	!snapshot.bullets.some(
		(bullet) =>
			bullet.enemy && predictedHit(bullet, snapshot, x, y, horizon)
	)

const friendInLine = (
	ai: { x: number; y: number },
	direction: TypeDirection,
	player: AiSnapshot['player']
) => {
	if (player.dead) return false

	const dx = player.x - ai.x
	const dy = player.y - ai.y
	if (direction === 'TOP') return Math.abs(dx) <= 7 && dy > 0
	if (direction === 'BOTTOM') return Math.abs(dx) <= 7 && dy < 0
	if (direction === 'RIGHT') return Math.abs(dy) <= 7 && dx > 0

	return Math.abs(dy) <= 7 && dx < 0
}

const shotPathClear = (
	ai: { x: number; y: number },
	direction: TypeDirection,
	target: { x: number; y: number },
	snapshot: AiSnapshot
) => {
	const distanceToTarget =
		direction === 'TOP' || direction === 'BOTTOM'
			? Math.abs(target.y - ai.y)
			: Math.abs(target.x - ai.x)
	const maxDistance = Math.min(distanceToTarget, 64)

	for (let step = 9; step <= maxDistance; step += 4) {
		const point = stepPosition(ai.x, ai.y, direction, step)
		for (const object of snapshot.obstacles) {
			if (!object.blocksBullets) continue
			if (
				point.x >= object.x - 2 &&
				point.x <= object.x + 9 &&
				point.y >= object.y - 2 &&
				point.y <= object.y + 9
			)
				return false
		}
	}

	return true
}

const firingDirection = (
	ai: { x: number; y: number },
	target: { x: number; y: number },
	tolerance: number
): TypeDirection | null => {
	const dx = target.x - ai.x
	const dy = target.y - ai.y
	if (Math.abs(dx) <= tolerance && Math.abs(dy) > tolerance)
		return dy > 0 ? 'TOP' : 'BOTTOM'
	if (Math.abs(dy) <= tolerance && Math.abs(dx) > tolerance)
		return dx > 0 ? 'RIGHT' : 'LEFT'

	return null
}

const bulletSpeedFor = (type: TypeTank) =>
	type === 'LVL_0' ? 1 : type === 'LVL_1' ? 2 : 3

const leadTarget = (
	ai: AiSnapshot['ai'],
	enemy: AiSnapshot['enemies'][number],
	features: AiTierFeatures
) => {
	if (!features.aimLead) return { x: enemy.x, y: enemy.y }

	const travel =
		distance(ai.x, ai.y, enemy.x, enemy.y) / bulletSpeedFor(ai.type)
	const lead = enemy.speed * travel

	return advance(enemy.x, enemy.y, enemy.direction, lead)
}

const adjacentBrick = (
	snapshot: AiSnapshot,
	direction: TypeDirection
): boolean => {
	const point = stepPosition(snapshot.ai.x, snapshot.ai.y, direction, 8)

	return snapshot.obstacles.some(
		(object) =>
			object.destructible &&
			point.x >= object.x - 2 &&
			point.x <= object.x + 9 &&
			point.y >= object.y - 2 &&
			point.y <= object.y + 9
	)
}

const drillDirection = (
	snapshot: AiSnapshot
): TypeDirection | null =>
	ALL_DIRECTIONS.find(
		(direction) =>
			adjacentBrick(snapshot, direction) &&
			!friendInLine(snapshot.ai, direction, snapshot.player)
	) ?? null

const chooseDirection = (
	ai: AiSnapshot['ai'],
	targetX: number,
	targetY: number,
	busy: BusyCoordinates[],
	player: AiSnapshot['player']
): TypeDirection | null => {
	const horizontal: TypeDirection = targetX >= ai.x ? 'RIGHT' : 'LEFT'
	const vertical: TypeDirection = targetY >= ai.y ? 'TOP' : 'BOTTOM'
	const options =
		Math.abs(targetX - ai.x) >= Math.abs(targetY - ai.y)
			? [horizontal, vertical]
			: [vertical, horizontal]

	for (const direction of options) {
		if (!canStep(ai, direction, busy)) continue
		const next = stepPosition(ai.x, ai.y, direction, 1)
		const blockingPlayer =
			!player.dead &&
			player.direction === direction &&
			Math.abs(next.x - player.x) <= 14 &&
			Math.abs(next.y - player.y) <= 14
		if (!blockingPlayer) return direction
	}

	for (const direction of [ai.direction, ...ALL_DIRECTIONS]) {
		if (canStep(ai, direction, busy)) return direction
	}

	return null
}

export const decideTactics = (
	snapshot: AiSnapshot,
	features: AiTierFeatures,
	role: AiRoleProfile,
	goal: AiGoal | null,
	path: LanePath | null,
	danger: DangerMap,
	memory: AiMemory,
	busy: BusyCoordinates[]
): AiDecision => {
	const ai = snapshot.ai
	const hold: AiDecision = { move: null, fire: false }

	if (features.dodgeHorizon > 0) {
		const incoming = findIncomingBullet(snapshot, features.dodgeHorizon)
		if (incoming) {
			const perpendicular: TypeDirection[] =
				incoming.direction === 'LEFT' || incoming.direction === 'RIGHT'
					? ['TOP', 'BOTTOM']
					: ['LEFT', 'RIGHT']
			const safe = perpendicular.filter((direction) => {
				if (!canStep(ai, direction, busy)) return false
				const next = stepPosition(ai.x, ai.y, direction, 1)

				return isCellSafe(snapshot, next.x, next.y, features.dodgeHorizon)
			})
			if (safe.length) {
				memory.path = null

				return { move: safe[0], fire: false }
			}

			if (
				features.blockBullets &&
				ai.availableBullets > 0 &&
				distance(ai.x, ai.y, incoming.x, incoming.y) <= 40
			) {
				const aim = firingDirection(ai, incoming, 7)
				if (
					aim &&
					ai.direction === aim &&
					!friendInLine(ai, aim, snapshot.player) &&
					shotPathClear(ai, aim, incoming, snapshot) &&
					memory.fireCooldown <= 0
				) {
					memory.fireCooldown = features.fireCooldown

					return { move: null, fire: true }
				}
			}

			if (features.escapeBrick && ai.availableBullets > 0) {
				const brick = drillDirection(snapshot)
				if (brick) {
					memory.path = null

					return ai.direction === brick
						? { move: null, fire: true }
						: { move: brick, fire: false }
				}
			}

			const retreat =
				incoming.direction === 'RIGHT'
					? 'LEFT'
					: incoming.direction === 'LEFT'
						? 'RIGHT'
						: incoming.direction === 'TOP'
							? 'BOTTOM'
							: 'TOP'
			if (canStep(ai, retreat, busy)) {
				memory.path = null

				return { move: retreat, fire: false }
			}
		}
	}

	if (goal?.enemyId) {
		const enemy = snapshot.enemies.find((e) => e.id === goal.enemyId)
		if (enemy) {
			const target = leadTarget(ai, enemy, features)
			const tolerance =
				features.aimTolerance + Math.random() * features.aimJitter
			const aim = firingDirection(ai, target, tolerance)
			const dist = distance(ai.x, ai.y, target.x, target.y)
			if (aim && dist <= features.engageRange) {
				if (ai.direction !== aim) return { move: aim, fire: false }
				if (
					memory.fireCooldown <= 0 &&
					ai.availableBullets > 0 &&
					!friendInLine(ai, aim, snapshot.player) &&
					shotPathClear(ai, aim, target, snapshot)
				) {
					memory.fireCooldown = features.fireCooldown

					return { move: null, fire: true }
				}
			}
		}
	}

	if (path?.brick && features.brickDrilling && ai.availableBullets > 0) {
		const direction = directionToWaypoint(ai, path.brick)
		if (
			direction &&
			distance(ai.x, ai.y, path.brick.x, path.brick.y) <= 20 &&
			adjacentBrick(snapshot, direction) &&
			!friendInLine(ai, direction, snapshot.player) &&
			memory.fireCooldown <= 0
		) {
			memory.fireCooldown = features.fireCooldown
			memory.path = null

			return ai.direction === direction
				? { move: null, fire: true }
				: { move: direction, fire: false }
		}
	}

	if (!goal) return hold
	if (
		!goal.enemyId &&
		distance(ai.x, ai.y, goal.x, goal.y) <= WAYPOINT_REACHED_DISTANCE
	)
		return hold

	let direction: TypeDirection | null = null
	if (features.pathfinding !== 'greedy' && path) {
		while (path.waypoints.length) {
			const waypoint = path.waypoints[0]
			if (waypoint.x === ai.x && waypoint.y === ai.y) {
				path.waypoints.shift()
				continue
			}
			const step = directionToWaypoint(ai, waypoint)
			if (!step) {
				path.waypoints.shift()
				continue
			}
			if (!canStep(ai, step, busy)) {
				memory.path = null
				break
			}
			direction = step
			break
		}
	}

	if (!direction)
		direction = chooseDirection(ai, goal.x, goal.y, busy, snapshot.player)

	return { move: direction, fire: false }
}
```

```ts
import { Game } from '../instances'
import { BusyCoordinates, TypeDirection } from '../types'
import { isPlayerCanMove, isPlayerDead } from '../utils'
import { AI_ROLE_PROFILES, AI_TIER_FEATURES } from './ai/ai.profiles'
import { buildSnapshot, createMemory } from './ai/ai.perception'
import { buildDangerMap, computeThreats } from './ai/threat'
import { findPath } from './ai/ai.path'
import { selectGoal } from './ai/ai.brain'
import { ALL_DIRECTIONS, canStep, decideTactics } from './ai/ai.tactics'
import { AiMemory } from './ai/ai.types'

const states = new WeakMap<Game, AiMemory>()
const STUCK_FRAME_LIMIT = 6

export const getAiMemory = (game: Game) => states.get(game)

export const aiFrameLogic = (game: Game) => {
	const { p1, p2, aiLevel } = game
	if (!p2 || !aiLevel) return

	const controls = game.p2Controls
	const features = AI_TIER_FEATURES[aiLevel]
	const role = AI_ROLE_PROFILES[game.aiRole ?? 'wingman']

	if (game.isPaused || !isPlayerCanMove(p2)) {
		controls.move = null
		controls.fire = false
		return
	}

	let memory = states.get(game)
	if (!memory) {
		memory = createMemory()
		states.set(game, memory)
	}

	const snapshot = buildSnapshot(game, memory)
	const busy: BusyCoordinates[] = [...game.enemies, ...game.objects]
	if (!isPlayerDead(p1)) busy.push(p1)

	if (memory.fireCooldown > 0) memory.fireCooldown--

	const danger = buildDangerMap(snapshot, Math.max(features.dodgeHorizon, 1))

	if (memory.cooldown <= 0 || !memory.goal) {
		const threats = computeThreats(snapshot)
		memory.goal = selectGoal(snapshot, threats, features, role, memory)
		memory.cooldown = features.decisionInterval
		memory.path = null
		memory.pathSignature = null
	}
	memory.cooldown--

	if (memory.goal && features.pathfinding !== 'greedy') {
		const needsPath =
			!memory.path || memory.pathSignature !== snapshot.mapSignature
		if (needsPath) {
			const ignoreId =
				memory.goal.enemyId ??
				(memory.goal.kind === 'patrol' || memory.goal.kind === 'support'
					? p1.id
					: undefined)
			memory.path = findPath(snapshot.ai, memory.goal.x, memory.goal.y, busy, {
				allowBrick: features.brickDrilling,
				danger: features.dodgeHorizon > 0 ? danger : undefined,
				dangerPenalty: 0.5,
				dangerHorizon: features.dodgeHorizon,
				ignoreId,
			})
			memory.pathSignature = snapshot.mapSignature
		}
	}

	const decision = decideTactics(
		snapshot,
		features,
		role,
		memory.goal,
		memory.path,
		danger,
		memory,
		busy
	)
	controls.move = decision.move
	controls.fire = decision.fire

	const moved =
		p2.coordinateX !== memory.lastX || p2.coordinateY !== memory.lastY
	if (!moved && memory.lastMove !== null && p2.direction === memory.lastMove) {
		memory.stuckFrames++
	} else {
		memory.stuckFrames = 0
		memory.stuckRecoveries = 0
	}

	if (memory.stuckFrames > STUCK_FRAME_LIMIT) {
		memory.stuckFrames = 0
		memory.stuckRecoveries++
		memory.path = null
		if (memory.stuckRecoveries >= 2) {
			memory.stuckRecoveries = 0
			memory.goal = null
			memory.cooldown = 0
		} else {
			const sidestep = ALL_DIRECTIONS.find((direction: TypeDirection) =>
				canStep(snapshot.ai, direction, busy)
			)
			if (sidestep) controls.move = sidestep
		}
	}

	memory.lastX = p2.coordinateX
	memory.lastY = p2.coordinateY
	memory.lastMove = controls.move
}
```

```ts
// @vitest-environment node
import { afterEach, describe, expect, it, vi } from 'vitest'
import { TypeAiLevel, TypeAiRole } from '../../types'
import { gameFrame } from '../index'
import { aiSettings, makeAiGame } from './ai.test-utils'

const FRAMES = 3000
const SEEDS = [1, 7, 42]

interface Metrics {
	score: number
	baseAlive: boolean
	stuckRatio: number
	friendlyFire: number
	nearBaseRatio: number
}

const run = (seed: number, level: TypeAiLevel, role: TypeAiRole): Metrics => {
	let value = seed
	vi.spyOn(Math, 'random').mockImplementation(() => {
		value |= 0
		value = (value + 0x6d2b79f5) | 0
		let t = Math.imul(value ^ (value >>> 15), 1 | value)
		t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
		return ((t ^ (t >>> 14)) >>> 0) / 4294967296
	})

	const game = makeAiGame(level, role, { ...aiSettings, friendlyFire: true })
	const seen = new Set<string>()
	let kills = 0
	let lastLevel = game.level
	let stuckFrames = 0
	let movingFrames = 0
	let nearBaseFrames = 0
	let friendlyFire = 0

	for (let i = 0; i < FRAMES; i++) {
		const p2 = game.p2!
		const beforeX = p2.coordinateX
		const beforeY = p2.coordinateY
		const p1LivesBefore = game.p1.lives
		const move = game.p2Controls.move

		gameFrame(game)

		if (game.level !== lastLevel) {
			seen.clear()
			lastLevel = game.level
		}
		for (const enemy of game.enemies) seen.add(enemy.id)
		for (const id of [...seen]) {
			if (!game.enemies.some((enemy) => enemy.id === id)) {
				seen.delete(id)
				kills++
			}
		}
		if (move !== null) {
			movingFrames++
			if (p2.coordinateX === beforeX && p2.coordinateY === beforeY)
				stuckFrames++
		}
		if (p2.coordinateY < 100) nearBaseFrames++
		if (game.p1.lives < p1LivesBefore) friendlyFire++

		if (!game.isFlagAlive || !p2.lives) break
	}

	const baseAlive = game.isFlagAlive
	const score = FRAMES + 50 * kills - 500 * (baseAlive ? 0 : 1)

	return {
		score,
		baseAlive,
		stuckRatio: movingFrames ? stuckFrames / movingFrames : 0,
		friendlyFire,
		nearBaseRatio: nearBaseFrames / FRAMES,
	}
}

const average = (values: number[]) =>
	values.reduce((sum, value) => sum + value, 0) / values.length

afterEach(() => vi.restoreAllMocks())

describe('AI soak benchmarks', () => {
	it('orders tiers by composite score', () => {
		const tiers: TypeAiLevel[] = [
			'beginner',
			'intermediate',
			'pro',
			'elite',
		]
		const averages = tiers.map((tier) =>
			average(SEEDS.map((seed) => run(seed, tier, 'wingman').score))
		)

		expect(averages[3]).toBeGreaterThan(averages[0])
		expect(averages[2]).toBeGreaterThan(averages[1])
		for (let i = 1; i < averages.length; i++) {
			expect(averages[i]).toBeGreaterThanOrEqual(averages[i - 1] - 400)
		}
	})

	it('keeps the guardian closer to the base than the hunter', () => {
		const guardian = average(
			SEEDS.map((seed) => run(seed, 'pro', 'guardian').nearBaseRatio)
		)
		const hunter = average(
			SEEDS.map((seed) => run(seed, 'pro', 'hunter').nearBaseRatio)
		)

		expect(guardian).toBeGreaterThan(hunter)
	})

	it('never friendly-fires, never wedges and never loses the base on elite', () => {
		for (const seed of SEEDS) {
			const metrics = run(seed, 'elite', 'wingman')
			expect(metrics.friendlyFire).toBe(0)
			expect(metrics.stuckRatio).toBeLessThan(0.05)
			expect(metrics.baseAlive).toBe(true)
		}
	})
})
```
